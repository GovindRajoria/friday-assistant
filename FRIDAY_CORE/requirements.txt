# The IANA timezone database, for world_time. Named explicitly rather than
# inherited: Linux and macOS have a system copy that zoneinfo reads, and Windows
# has none at all — ZoneInfo("Asia/Tokyo") raises there unless this is installed.
# It arrived transitively on the development machine, which is exactly how a
# dependency ends up working here and nowhere else.
tzdata

# Reading skills (5a). pypdf for read_document, openpyxl for read_spreadsheet,
# icalendar for the calendar skill's .ics parsing. All three are imported inside
# the method that needs them, never at module scope — core/registry.py drops a
# skill whose import raises, silently, so a module-scope import here would make
# a missing package look like a skill that never existed.
pypdf
openpyxl
icalendar

# ocr_screen reads characters off the screen through the OCR engine already in
